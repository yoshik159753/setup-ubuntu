= README

[source,sh]
.usage
----
ansible-playbook -i localhost, --connection local main.yml --ask-become-pass

# 途中から実行
ansible-playbook -i localhost, --connection local main.yml --ask-become-pass --start-at='rolename : taskname'

# 利用可能なタグを確認
ansible-playbook -i localhost, --connection local main.yml --list-tags

# 特定のタグのみ実行
ansible-playbook -i localhost, --connection local main.yml --ask-become-pass --tags git

# 複数のタグを指定
ansible-playbook -i localhost, --connection local main.yml --ask-become-pass --tags "git,nodejs,docker"
----

[source,sh]
.ansible install
----
# ubuntu 23.04 の場合は初期状態でも最新のバージョンがインストールできる模様
sudo apt-get update && sudo apt-get install -y ansible

# 古いバージョンでは ppa を追加すると良い
# ref. http://docs.ansible.com/ansible/intro_installation.html
sudo apt-get update && sudo apt-get install -y software-properties-common
sudo apt-add-repository ppa:ansible/ansible
sudo apt-get update && sudo apt-get install -y ansible
----
